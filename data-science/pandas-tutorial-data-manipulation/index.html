Example 1 · Baseline example
# Pandas Tutorial · example 1
import pandas as pd
orders = pd.read_csv('orders.csv')
customers = pd.read_csv('customers.csv')
merged = orders.merge(customers, on='customer_id', how='left', validate='many_to_one')
summary = (merged[merged['status'].eq('complete')]
           .groupby('region')
           .agg(revenue=('amount','sum'), orders=('order_id','nunique')))
print(summary)
