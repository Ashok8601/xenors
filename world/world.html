import numpy as np

def evaluate_market_friction(liquidity_vector, threshold=0.85):
    """
    Decodes real-time server order book velocity.
    """
    matrix = np.array(liquidity_vector)
    normalized_score = np.tanh(np.mean(matrix))
    
    if normalized_score > threshold:
        return "HIGH_CONVERGENCE_TRIGGER"
    return "STABLE_HOLD"
